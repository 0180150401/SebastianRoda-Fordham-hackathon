    ```ts
    import * as z from "zod";

    export const STEP_IDS = ["sources", "synthesis", "images"] as const;
    export type StepId = (typeof STEP_IDS)[number];

    export const STEP_STATUSES = ["running", "done", "error"] as const;
    export type StepStatus = (typeof STEP_STATUSES)[number];

    // Hard cap to mitigate DoS via oversized lines (see threat_model T-01-02).
    // Consumers should refuse to buffer a single NDJSON line larger than this.
    export const MAX_NDJSON_LINE_BYTES = 1_000_000; // 1 MB

    const stepEventSchema = z.object({
      type: z.literal("step"),
      id: z.enum(STEP_IDS),
      status: z.enum(STEP_STATUSES),
      detail: z.string().max(2_000).optional(),
    });

    const doneEventSchema = z.object({
      type: z.literal("done"),
      // Envelope-only validation per RESEARCH Pitfall 1: payload is normalized via
      // lib/pipeline/normalize-payload.ts before reaching FSM consumers, so we keep
      // the inner shape as `unknown` here and let the normalizer enforce structure.
      // Plan 02 contract tests cover the envelope; Plan 03 wires normalization.
      payload: z.unknown(),
    });

    const errorEventSchema = z.object({
      type: z.literal("error"),
      message: z.string().min(1).max(4_000),
    });

    export const pipelineEventSchema = z.discriminatedUnion("type", [
      stepEventSchema,
      doneEventSchema,
      errorEventSchema,
    ]);

    export type PipelineEvent = z.infer<typeof pipelineEventSchema>;
    export type StepEvent  = z.infer<typeof stepEventSchema>;
    export type DoneEvent  = z.infer<typeof doneEventSchema>;
    export type ErrorEvent = z.infer<typeof errorEventSchema>;
    ```

    Notes:
    - `MAX_NDJSON_LINE_BYTES` exported here (not the parser) so the schema module owns the contract; Plan 02's parser imports and enforces it.
    - Discriminator key MUST stay `"type"` to match server emission verbatim (per D-02 / RESEARCH).
    - Do NOT import anything from `next/*` or DOM types — this module is shared with Vitest in Node and must stay pure.
  </action>
