    ```ts
    import { pipelineEventSchema, MAX_NDJSON_LINE_BYTES, type PipelineEvent } from "@/lib/pipeline/types";

    export class NdjsonParseError extends Error {
      readonly cause?: unknown;
      readonly rawLineSnippet?: string;
      constructor(message: string, opts?: { cause?: unknown; rawLineSnippet?: string }) {
        super(message);
        this.name = "NdjsonParseError";
        this.cause = opts?.cause;
        // Snippet for debugging only — capped to 200 chars to avoid leaking large payloads in error UI.
        this.rawLineSnippet = opts?.rawLineSnippet?.slice(0, 200);
      }
    }

    // Streaming parser — used by the hook (Plan 03) where chunks arrive over time.
    export function createNdjsonParser() {
      let buffer = "";
      function pushChunk(chunk: string): PipelineEvent[] {
        if (buffer.length + chunk.length > MAX_NDJSON_LINE_BYTES * 2) {
          // Soft cap on accumulated unflushed bytes (defends against a single huge line).
          throw new NdjsonParseError("Stream line exceeded maximum size");
        }
        buffer += chunk;
        const out: PipelineEvent[] = [];
        let nl: number;
        while ((nl = buffer.indexOf("\n")) >= 0) {
          const line = buffer.slice(0, nl);
          buffer = buffer.slice(nl + 1);
          const trimmed = line.trim();
          if (trimmed === "") continue;
          if (Buffer.byteLength(trimmed, "utf8") > MAX_NDJSON_LINE_BYTES) {
            throw new NdjsonParseError("NDJSON line exceeded MAX_NDJSON_LINE_BYTES");
          }
          let json: unknown;
          try { json = JSON.parse(trimmed); }
          catch (e) {
            throw new NdjsonParseError("Stream contained invalid JSON", { cause: e, rawLineSnippet: trimmed });
          }
          const result = pipelineEventSchema.safeParse(json);
          if (!result.success) {
            const issue = result.error.issues[0];
            const path = issue?.path?.join(".") || "<root>";
            throw new NdjsonParseError(`Stream event failed validation at "${path}": ${issue?.message ?? "unknown"}`, { cause: result.error });
          }
          out.push(result.data);
        }
        return out;
      }
      function end(): PipelineEvent[] {
        // Per Pitfall 2: trailing non-empty content at EOF is fatal under D-01.
        if (buffer.trim() !== "") {
          const snippet = buffer; buffer = "";
          throw new NdjsonParseError("Stream ended with incomplete line", { rawLineSnippet: snippet });
        }
        buffer = "";
        return [];
      }
      return { pushChunk, end };
    }

    // Convenience for tests + any caller with the whole stream in memory.
    export function parseNdjsonEvents(chunks: Iterable<string>): PipelineEvent[] {
      const parser = createNdjsonParser();
      const events: PipelineEvent[] = [];
      for (const c of chunks) events.push(...parser.pushChunk(c));
      events.push(...parser.end());
      return events;
    }
    ```

## STRIDE Threat Register (ASVS L1, V5 Input Validation)

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-01 | Tampering | NDJSON line injection (malicious or corrupted JSON between valid events) | mitigate | `parseNdjsonEvents` throws `NdjsonParseError` on first invalid line; no skip/continue (D-01 + Pattern 3 in RESEARCH) |
| T-02-02 | Denial of Service | Single oversize line (memory exhaustion) | mitigate | Hard byte cap `MAX_NDJSON_LINE_BYTES` enforced before `JSON.parse`; soft cap on accumulated buffer |
| T-02-03 | Denial of Service | Stream never produces "\n" (unbounded buffer growth) | mitigate | Soft cap of `2 × MAX_NDJSON_LINE_BYTES` accumulated bytes triggers `NdjsonParseError` |
| T-02-04 | Information Disclosure | Error messages echoing raw event content into UI | mitigate | `NdjsonParseError.message` only references issue `path` + zod `issue.message`; raw content stored on `rawLineSnippet` capped at 200 chars; parser never `console.log`s raw lines |
| T-02-05 | Tampering | Schema drift between server `emitLine` literals and client schema | mitigate | Task 3 types `emitLine: (..., event: PipelineEvent)` so any drift is a TS error; "Server-shape parity" tests in Task 2 give a runtime backstop |
| T-02-06 | Repudiation | (n/a) | accept | Audit logging is Phase 2's concern |
