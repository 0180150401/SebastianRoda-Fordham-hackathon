    ```ts
    "use client";

    import { useCallback, useRef } from "react";
    import { createNdjsonParser, NdjsonParseError } from "@/lib/pipeline/ndjson";
    import type { PipelineEvent, StepEvent, DoneEvent, ErrorEvent } from "@/lib/pipeline/types";

    export type StreamCallbacks = {
      onStep?: (event: StepEvent) => void;
      onDone?: (event: DoneEvent) => void;
      // Server-emitted error event (run reported a failure but the wire was valid).
      onError?: (event: ErrorEvent) => void;
      // Fatal parser/validator failure (D-01) — wire was malformed or schema-invalid.
      // Caller surfaces a user-legible message via its existing setLoadError path.
      onFatal?: (error: NdjsonParseError | Error) => void;
    };

    export function useSemanticUniverseStream() {
      // useRef to keep callback identity stable across renders without re-running the consumer.
      const consume = useCallback(async (
        body: ReadableStream<Uint8Array>,
        callbacks: StreamCallbacks,
        signal?: AbortSignal,
      ): Promise<void> => {
        const reader = body.getReader();
        const decoder = new TextDecoder();
        const parser = createNdjsonParser();

        const dispatch = (events: PipelineEvent[]) => {
          for (const ev of events) {
            if (ev.type === "step")  callbacks.onStep?.(ev);
            else if (ev.type === "done")  callbacks.onDone?.(ev);
            else if (ev.type === "error") callbacks.onError?.(ev);
          }
        };

        try {
          // eslint-disable-next-line no-constant-condition
          while (true) {
            if (signal?.aborted) throw new DOMException("aborted", "AbortError");
            const { done, value } = await reader.read();
            if (done) break;
            const chunk = decoder.decode(value, { stream: true });
            dispatch(parser.pushChunk(chunk));
          }
          // Flush trailing decoder bytes; treat non-empty trailing buffer as fatal (D-01).
          dispatch(parser.pushChunk(decoder.decode()));
          dispatch(parser.end());
        } catch (err) {
          callbacks.onFatal?.(err instanceof Error ? err : new Error(String(err)));
          throw err; // re-throw so caller's existing try/catch path runs (setLoadError, fallback graph).
        } finally {
          try { reader.releaseLock(); } catch { /* reader may already be released */ }
        }
      }, []);
      return { consume };
    }
    ```
